import pandas as pd
import numpy as np
import matplotlib.pyplot as plt

# Convert 'Created Date' and 'Closed Date' to datetime
data['Created Date'] = pd.to_datetime(data['Created Date'], errors='coerce')
data['Closed Date'] = pd.to_datetime(data['Closed Date'], errors='coerce')

# Calculate the 'Request_Closing_Time' (time taken to close the request in hours)
data['Request_Closing_Time'] = (data['Closed Date'] - data['Created Date']).dt.total_seconds() / 3600  # in hours

# Group by 'Complaint Type' and 'Borough', calculate average closing time
pivot_table = data.groupby(['Complaint Type', 'Borough'])['Request_Closing_Time'].mean().reset_index()

# Get top 5 most common complaint types
top_complaints = data['Complaint Type'].value_counts().nlargest(5).index

# Filter to include only top complaint types
pivot_top = pivot_table[pivot_table['Complaint Type'].isin(top_complaints)]

# Plot setup
plt.figure(figsize=(14, 8))
bar_width = 0.15
complaints = top_complaints.tolist()
x = np.arange(len(complaints))

boroughs = pivot_top['Borough'].unique()
colors = plt.cm.Set2(np.linspace(0, 1, len(boroughs)))

# Loop through each borough and plot
for i, borough in enumerate(boroughs):
    borough_data = pivot_top[pivot_top['Borough'] == borough]
    heights = []
    for complaint in complaints:
        value = borough_data.loc[borough_data['Complaint Type'] == complaint, 'Request_Closing_Time']
        heights.append(value.values[0] if not value.empty else 0)
    
    plt.bar(x + i * bar_width, heights, width=bar_width, label=borough, color=colors[i])

# Labels and layout
plt.title('Average Request Closing Time by Complaint Type and Borough')
plt.xlabel('Complaint Type')
plt.ylabel('Average Closing Time (Hours)')
plt.xticks(x + (bar_width * (len(boroughs) - 1) / 2), complaints, rotation=45)
plt.legend(title='Borough')
plt.tight_layout()

# Show plot
plt.show()

import scipy.stats as stats 
# Convert 'Created Date' and 'Closed Date' to datetime
data['Created Date'] = pd.to_datetime(data['Created Date'], errors='coerce')
data['Closed Date'] = pd.to_datetime(data['Closed Date'], errors='coerce')

# Calculate the 'Request_Closing_Time' (time taken to close the request in hours)
data['Request_Closing_Time'] = (data['Closed Date'] - data['Created Date']).dt.total_seconds() / 3600  # in hours

# Drop rows with NaN values in 'Request_Closing_Time' and 'Complaint Type'
data_clean = data.dropna(subset=['Request_Closing_Time', 'Complaint Type'])

# Perform One-Way ANOVA on 'Request_Closing_Time' across 'Complaint Type'
complaint_types = data_clean['Complaint Type'].unique()

# Grouping the data by 'Complaint Type' and collecting 'Request_Closing_Time' for each complaint type
grouped_data = [data_clean[data_clean['Complaint Type'] == complaint]['Request_Closing_Time'] for complaint in complaint_types]

# Perform the ANOVA test
f_stat, p_value = stats.f_oneway(*grouped_data)

# Print the result
print(f"ANOVA F-statistic: {f_stat}")
print(f"p-value: {p_value}")

# Interpretation of the results
if p_value < 0.05:
    print("We reject the null hypothesis. The average response time across complaint types is significantly different.")
else:
    print("We fail to reject the null hypothesis. There is no significant difference in the average response time across complaint types.")
